
#include <stdio.h>

int main() {
    char nome[50], endereco[100], telefone[12];
    int pessoas, opcao;
    FILE *arquivo;

    do {
        printf("\n===== SISTEMA SOCIAL =====\n");
        printf("1 - Cadastrar familia\n");
        printf("2 - Consultar cadastros\n");
        printf("0 - Sair\n");
        printf("Escolha: ");
        scanf("%d", &opcao);

        if (opcao == 1) {
            printf("\n=== CADASTRO DE FAMILIAS ===\n");

            printf("Nome do responsavel: ");
            scanf(" %49[^\n]", nome);

            printf("Endereco: ");
            scanf(" %99[^\n]", endereco);

            printf("Telefone (11 digitos): ");
            scanf(" %11[^\n]", telefone);

            printf("Quantidade de pessoas: ");
            scanf("%d", &pessoas);

            arquivo = fopen("familias.txt", "a");

            if (arquivo == NULL) {
                printf("Erro ao salvar cadastro!\n");
                return 1;
            }

            fprintf(arquivo, "Nome: %s\n", nome);
            fprintf(arquivo, "Endereco: %s\n", endereco);
            fprintf(arquivo, "Telefone: %s\n", telefone);
            fprintf(arquivo, "Pessoas: %d\n", pessoas);
            fprintf(arquivo, "------------------------\n");

            fclose(arquivo);

            printf("\nFamilia cadastrada com sucesso!\n");
        }

        else if (opcao == 2) {
            char linha[150];
            int encontrou = 0;

            arquivo = fopen("familias.txt", "r");

            if (arquivo == NULL) {
                printf("\nNenhuma familia cadastrada!\n");
            } else {
                printf("\n===== FAMILIAS CADASTRADAS =====\n");

                while (fgets(linha, sizeof(linha), arquivo) != NULL) {
                    printf("%s", linha);
                    encontrou = 1;
                }

                if (!encontrou)
                    printf("Nenhum cadastro encontrado!\n");

                fclose(arquivo);
            }
        }

        else if (opcao != 0) {
            printf("\nOpcao invalida!\n");
        }

    } while (opcao != 0);

    printf("\nSistema encerrado!\n");

    return 0;
}
